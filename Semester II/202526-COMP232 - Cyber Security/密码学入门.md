# Hamming Distance

两个字符串`a`和`b`首先转化为`byte[]`数组，数组长度必须一致

```java
public static int hammingDistance(byte[] bytesA, byte[] bytesB) {
    if (bytesA.length != bytesB.length) {
        throw new IllegalArgumentException("Strings must have equal length");
    }
    int distance = 0;
    for (int i = 0; i < bytesA.length; i++) {
        distance += Integer.bitCount((bytesA[i] ^ bytesB[i]) & 0xFF);
    }
    return distance;
}
```



`byte`类型的范围是 $[-128,127]$，也就是 `0x00` 到 `0xFF` 一共8个比特位（1个`F`是4比特位），$2^{8}=256$。

> [!CAUTION]
>
> 在Java中，所有的`byte` XOR操作过程都会升级到`int`，如果`byte`为负数，上升为`int`后其最高位会补1，这回对Hamming Distance有勿扰，比如正负比较的情况，因此，必须加以掩码`0xFF`以确保最高位被掩盖掉！

# 评估句子

该函数用于评估一个句子是否“长得像一个英文句子”

```java
public static double evaluateSentenceScore(String text) {
    double score = 0;
    String tally = "etaoin SHRDLU";
    for (char c : text.toLowerCase().toCharArray())
        if (Character.isLetter(c) || c == ' ') {
            score += 1;     // 如果是正常英文或者空格，加1分
            if (tally.indexOf(c) != -1)
                score += 2; // 在常驻字母表中，加2分
        } else if (Character.isISOControl(c))
            score -= 10;    // 乱码或者控制字符，减10分
    return score;
}
```

# Hex 转 Byte

这里说的 Hex 一般指的是，Hex 字符串

- 例如：`aaff1c`
- 可以看作`byte[]`：`0xaa`，`0xff`，`0x1c`



```java
public static byte[] hexToByte(String s) {
    int len = s.length();
    byte[] data = new byte[len / 2];
    for (int i = 0; i < len; i += 2)
        data[i / 2] = (byte) ((Character.digit(s.charAt(i), 16) << 4) |
                              Character.digit(s.charAt(i + 1), 16));
    return data;
}
```



在这个函数中

1. 返回的`byte`一定是`String s`长度的一半
2. 使用`|`操作可以把高位和低位直接合并，通过左移`<<`



# Byte 转 Hex

这个功能用的并不多，和上面的 **Hex 转 Byte** 恰好相反，也就是把`byte[]`：`0xaa`，`0xff`，`0x1c`转为`aaff1c`字符串



```java
public static String byteToHex(byte[] bytes) {
    StringBuilder sb = new StringBuilder();
    for (byte b : bytes)
        sb.append(String.format("%02x", b & 0xFF));
    return sb.toString();
}
```



在这个函数中

1. sb的长度一定是bytes长度的2倍
2. `"%02x"`中`x`代表十六进制，`0`代表不足前面补0，`2`代表必须**至少**输出两位
3. 如果是`X`那么就是输出`AA`，如果是`x`那么就是输出`aa`



# 实现 RSA 加密



```java
import javax.crypto.Cipher;
import javax.crypto.spec.OAEPParameterSpec;
import javax.crypto.spec.PSource;
import java.io.*;
import java.math.BigInteger;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.security.Key;
import java.security.KeyFactory;
import java.security.SecureRandom;
import java.security.spec.MGF1ParameterSpec;
import java.security.spec.RSAPrivateKeySpec;
import java.security.spec.RSAPublicKeySpec;

public class RSA implements Serializable {
    // 建议增加一个版本 ID，确保序列化兼容性
    private static final long serialVersionUID = 1L;

    private static final BigInteger E = new BigInteger("65537", 10); // 公钥指数
    private final BigInteger D; // 私钥
    private BigInteger N; // 模数
    private transient   Cipher encryptCipher;
    private transient Cipher decryptCipher;
    private final Key publicKey;
    private final Key privateKey;
    private RSA(int lengthOfKey) throws Exception {
        SecureRandom random = new SecureRandom();
        int bitLength = lengthOfKey / 2;

        // probability of being prime is (1 - 1/2^100)
        while (true) {
            BigInteger T; // 公钥指数

            BigInteger p = new BigInteger(bitLength, 100, random);
            BigInteger q = new BigInteger(bitLength, 100, random);

            // 1. 确保 p 和 q 不相等
            if (p.equals(q)) continue;

            this.N = p.multiply(q);

            // 2. 严谨起见，确保 N 的位数正确（有时两个高位较小的数相乘会导致位数少一位）
            if (N.bitLength() != lengthOfKey) continue;

            // 3. 设置欧拉函数
            T = p.subtract(BigInteger.ONE).multiply(q.subtract(BigInteger.ONE));

            // 4. 选择E，尝试 65537，如果 T<= 65537 那么重新尝试
            if (T.compareTo(E) <= 0) continue;

            // 5. 如果 E 是 T 的因子， 重试
            if (!T.gcd(E).equals(BigInteger.ONE)) continue;

            // 6. 计算私钥 D 使得 (D * E) % T == 1
            this.D = E.modInverse(T);

            // 7. 设置公钥和其他参数
            this.publicKey = KeyFactory.getInstance("RSA").generatePublic(new RSAPublicKeySpec(this.N, E));
            this.privateKey = KeyFactory.getInstance("RSA").generatePrivate(new RSAPrivateKeySpec(this.N, D));

            OAEPParameterSpec oaepParams = new OAEPParameterSpec("SHA-256", "MGF1", MGF1ParameterSpec.SHA256, PSource.PSpecified.DEFAULT);

            encryptCipher = Cipher.getInstance("RSA/ECB/OAEPWithSHA-256AndMGF1Padding");
            encryptCipher.init(Cipher.ENCRYPT_MODE, publicKey, oaepParams);

            decryptCipher = Cipher.getInstance("RSA/ECB/OAEPWithSHA-256AndMGF1Padding");
            decryptCipher.init(Cipher.DECRYPT_MODE, privateKey, oaepParams);
            break;
        }
    }
    public static RSA load(String path) throws Exception {
        try (ObjectInputStream ois = new ObjectInputStream(Files.newInputStream(Paths.get(path)))) {
            RSA loadedRSA = (RSA) ois.readObject();
            // 关键步骤：重新初始化 Cipher 对象
            loadedRSA.reinitializeCiphers();
            System.out.println("RSA 对象已成功从 " + path + " 加载");
            return loadedRSA;
        }
    }
    // 辅助方法：由于 Cipher 没被保存，回来后需要重新初始化
    private void reinitializeCiphers() throws Exception {
        OAEPParameterSpec oaepParams = new OAEPParameterSpec(
                "SHA-256", "MGF1", MGF1ParameterSpec.SHA256, PSource.PSpecified.DEFAULT);

        this.encryptCipher = Cipher.getInstance("RSA/ECB/OAEPWithSHA-256AndMGF1Padding");
        this.encryptCipher.init(Cipher.ENCRYPT_MODE, this.publicKey, oaepParams);

        this.decryptCipher = Cipher.getInstance("RSA/ECB/OAEPWithSHA-256AndMGF1Padding");
        this.decryptCipher.init(Cipher.DECRYPT_MODE, this.privateKey, oaepParams);
    }

    private byte[] encrypt(byte[] data) throws Exception {
        return encrypt(data,true);
    }
    private byte[] encrypt(byte[] data, boolean terminalOutputProcess) throws Exception {
        final int MAX_PAYLOAD = 190;
        // 使用动态获取输出长度，确保 1024/2048/4096 位密钥通用
        final int OUTPUT_BLOCK_SIZE = this.N.bitLength() / 8;

        int SELECT_BLOCK_SIZE = (int) Math.ceil((double) data.length / MAX_PAYLOAD);
        byte[] result = new byte[SELECT_BLOCK_SIZE * OUTPUT_BLOCK_SIZE];

        if (terminalOutputProcess) System.out.printf("Total Blocks = %d\n", SELECT_BLOCK_SIZE);

        for (int i = 0; i < SELECT_BLOCK_SIZE; i++) {
            if (terminalOutputProcess) {
                System.out.printf("\rProcessing: %.2f%%", (i * 100.0 / SELECT_BLOCK_SIZE));
            }

            int offset = i * MAX_PAYLOAD;
            // 关键修复：计算实际需要读取的长度，防止越界
            int lengthToRead = Math.min(MAX_PAYLOAD, data.length - offset);

            // 关键优化：直接从 data 加密，无需中间数组 temporary
            byte[] blockResult = encryptCipher.doFinal(data, offset, lengthToRead);

            // 写入结果数组
            System.arraycopy(blockResult, 0, result, i * OUTPUT_BLOCK_SIZE, blockResult.length);
        }

        if (terminalOutputProcess) System.out.println("\rProcessing: 100.00% ... Done!");
        return result;
    }

    private byte[] decrypt(byte[] data) throws Exception {
        return decrypt(data,true);
    }
    private byte[] decrypt(byte[] data, boolean terminalOutputProcess) throws Exception {
        final int OUTPUT_BLOCK_SIZE = this.N.bitLength() / 8;
        final int MAX_PAYLOAD = 190;
        int expectedBlockSize = data.length / OUTPUT_BLOCK_SIZE;
        byte[] result = new byte[expectedBlockSize*MAX_PAYLOAD];
        int resultIndex = 0;
        if (terminalOutputProcess) System.out.printf("Total Blocks = %d\n", expectedBlockSize);
        for (int i = 0; i < expectedBlockSize; i++) {
            if (terminalOutputProcess) {
                System.out.printf("\rProcessing: %.2f%%", (i * 100.0 / expectedBlockSize));
            }
            byte[] block = decryptCipher.doFinal(data, i * OUTPUT_BLOCK_SIZE, OUTPUT_BLOCK_SIZE);
            System.arraycopy(block,0,result,resultIndex,block.length);
            resultIndex+=block.length;
        }
        byte[] realResult = new byte[resultIndex];
        System.arraycopy(result,0,realResult,0,realResult.length);
        return realResult;
    }

    public static void saveAndGetInstance(String path) throws IOException {
        try (ObjectOutputStream oos = new ObjectOutputStream(Files.newOutputStream(Paths.get(path)))) {
            RSA rsa = new RSA(1024);
            oos.writeObject(rsa);
            System.out.println("RSA 对象已安全保存至: " + path);
        } catch (Exception e) {
            System.err.println("Cannot get RSA instance! ".concat(e.getMessage()));
        }
    }


    public static void encrypt(String filePath, RSA rsa) throws Exception {
        File inputFile = new File(filePath);
        File outputFile = new File(inputFile.getParent(), inputFile.getName().concat(".encrypt"));
        byte[] bytes = Files.readAllBytes(Paths.get(inputFile.getAbsolutePath()));
        byte[] encrypt = rsa.encrypt(bytes);
        Files.write(Paths.get(outputFile.getPath()),encrypt);
    }

    public static void decrypt(String filePath, RSA rsa) throws Exception{
        File inputFile = new File(filePath);
        File outputFile = new File(inputFile.getParent(), inputFile.getName().replace(".encrypt","1"));
        byte[] bytes = Files.readAllBytes(Paths.get(inputFile.getAbsolutePath()));
        byte[] decrypt = rsa.decrypt(bytes);
        Files.write(Paths.get(outputFile.getPath()),decrypt);
    }
}
```

    RSA rsa = RSA.load("/home/robot/Downloads/RSA_FILE");
