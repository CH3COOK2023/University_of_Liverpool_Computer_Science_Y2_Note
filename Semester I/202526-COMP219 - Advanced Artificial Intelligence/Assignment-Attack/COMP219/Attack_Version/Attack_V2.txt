def your_attack(model, X, y, device):

    epsilon = 0.1   # epsilon根据题目限制在 < 0.11
    num_steps = 40
    alpha = 0.015  # 步长
    num_classes = 10

    model.eval()

    X_orig = X.detach().clone().to(device)
    y = y.to(device)
    batch_size = X_orig.size(0)

    # 初始化最佳攻击结果
    # best_adv 存储目前为止让模型错得最离谱的图片
    # best_margin 存储这种错误的确信度（越大说明攻击越成功）
    best_adv = X_orig.clone()
    best_margin = torch.full((batch_size,), -1e9, device=device)

    # --- 阶段 1: 遍历每一个目标类别 (Targeted Attack) ---
    # FashionMNIST 有 0-9 共 10 类。我们尝试把图片伪装成除了 y 以外的所有类。
    for target_class in range(num_classes):

        # 创建目标标签向量
        target_labels = torch.full((batch_size,), target_class, device=device, dtype=torch.long)

        # 如果目标类别就是真实类别，没必要攻击，标记一下稍后跳过计算
        # (通过掩码 mask 控制，只攻击 target != true 的样本)
        target_neq_true = (target_labels != y)

        if target_neq_true.sum() == 0:
            continue

        # --- 随机初始化 (Random Start) ---
        X_adv = X_orig + torch.zeros_like(X_orig).uniform_(-epsilon, epsilon).to(device)
        X_adv = torch.clamp(X_adv, 0.0, 1.0)

        # --- PGD 迭代 ---
        for _ in range(num_steps):
            X_adv.requires_grad = True
            with torch.enable_grad():
                output = model(X_adv)

                # --- Targeted CW Loss ---
                # 我们希望：Target Class 的得分 (logit) 越高越好
                # 同时：Other Classes (包括真实类别) 的得分越低越好

                # 获取我们想要伪装成的 target_class 的得分
                target_logits = output.gather(1, target_labels.view(-1, 1)).squeeze(1)

                # 获取除了 target_class 以外最高的得分 (Max of Others)
                # 这代表了阻碍我们伪装的最大阻力
                one_hot_target = F.one_hot(target_labels, num_classes=num_classes).bool()
                other_logits = output.masked_fill(one_hot_target, -1e9)
                max_other_logits, _ = other_logits.max(dim=1)

                # Loss 定义：最大化 (Target - Max_Others)
                # 也就是说，我们优化 loss 使得 Target 得分超过其他所有得分
                # 注意：这里我们通常最小化 -(Target - Max_Others)，即最大化 Target - Max_Others
                loss = -(target_logits - max_other_logits).mean()

            # 计算梯度
            grad = torch.autograd.grad(loss, X_adv)[0]

            # 只有 target != true 的样本需要更新
            # 我们通过这种方式避免 shape mismatch，虽然全算也行，但这样更严谨
            grad[~target_neq_true] = 0

            # 更新对抗样本 (Targeted 是让 Loss 变小，因为我们定义的是负差值)
            # 或者理解为：我们想让 target logit 变大，所以沿着梯度负方向走（如果用的是标准CE）
            # 但这里我们显式写了 loss = -diff，所以我们 minimize loss，即沿着 -grad 方向
            # 等等，为避免混淆，最直观的写法是：
            # 我们想要 target_logits 变大 -> 对 target_logits 求导梯度是正的
            # 我们想要 max_other 变小 -> 对 max_other 求导梯度是负的
            # 所以直接： X_adv = X_adv - alpha * grad.sign()
            X_adv = X_adv.detach() - alpha * torch.sign(grad.detach())

            # 投影回 epsilon 球
            delta = torch.clamp(X_adv - X_orig, min=-epsilon, max=epsilon)
            X_adv = torch.clamp(X_orig + delta, min=0.0, max=1.0)

        # --- 记录最佳结果 ---
        with torch.no_grad():
            final_output = model(X_adv)

            # 计算攻击效果：Margin = Max_Non_True - True_Logit
            # 如果 Margin > 0，说明最强的非真实类得分 超过了 真实类得分 -> 攻击成功
            # 我们保留 Margin 最大的那个对抗样本 (错得最离谱的)

            # 1. 拿到真实类别的得分
            true_logits = final_output.gather(1, y.view(-1, 1)).squeeze(1)

            # 2. 拿到除了真实类别外最高的得分
            one_hot_true = F.one_hot(y, num_classes=num_classes).bool()
            other_logits_final = final_output.masked_fill(one_hot_true, -1e9)
            max_wrong_logits, _ = other_logits_final.max(dim=1)

            # 3. 计算 Margin
            current_margin = max_wrong_logits - true_logits

            # 4. 更新全局最佳
            # 只有在当前这轮 target attack 产生了更强的误导效果时，才更新 best_adv
            # 这里的 mask 确保我们只考虑当前 target 不等于 true 的情况
            update_mask = (current_margin > best_margin) & target_neq_true

            best_adv[update_mask] = X_adv[update_mask]
            best_margin[update_mask] = current_margin[update_mask]

    return best_adv.detach()
